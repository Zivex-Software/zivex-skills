// ---- Project and module detection (shared by the dotnet-* scripts; heuristic, same rules everywhere) ----
// A module is found by, in order: modules.json at the repo root ({"modules": {"Orders": ["src/Orders", "tests/Orders.Tests"]}, "shared": ["src/Shared"]}),
// the folder after "Modules/", or the project name minus a role suffix (Acme.Orders.Contracts -> Orders) and the common company prefix.
import { existsSync, readdirSync, readFileSync } from "node:fs";
import { basename, dirname, join, relative, resolve } from "node:path";

const SKIP_DIRS = new Set(["bin", "obj", "node_modules", ".git", ".vs", "TestResults", "artifacts"]);
const ROLE_SUFFIX = /\.(Contracts|Tests|UnitTests|IntegrationTests|Infrastructure|Domain|Application|Api|Endpoints|Migrations)$/i;
const SHARED_NAME = /(^|\.)(Shared|SharedKernel|BuildingBlocks|Common|Kernel)(\.|$)/i;
// Composition roots reference every module's implementation by design, so they belong to no module.
const HOST_NAME = /(^|\.)(Host|WebHost|AppHost|Bootstrapper)(\.|$)/i;
// Vendor/framework namespace roots: never a module, even when a module shares a segment name (Microsoft.AspNetCore.Identity vs module Identity).
const VENDOR_ROOTS = new Set(["System", "Microsoft", "Azure", "Amazon", "Google", "Newtonsoft", "Serilog", "MassTransit", "MediatR", "FluentValidation", "Xunit", "NUnit", "Moq", "NSubstitute", "Dapper", "Polly", "OpenTelemetry"]);
const posix = (p) => p.split("\\").join("/");

function* walkFiles(dir, depth = 0) {
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    if (e.isDirectory()) {
      if (!SKIP_DIRS.has(e.name) && depth < 24) yield* walkFiles(join(dir, e.name), depth + 1);
    } else yield join(dir, e.name);
  }
}

// Target framework set in the nearest Directory.Build.props above the project (stops at the repo root).
function inheritedTfm(file, root) {
  for (let d = dirname(file); ; d = dirname(d)) {
    const t = tryRead(join(d, "Directory.Build.props"));
    const m = t && /<TargetFrameworks?>([^<$]+)</i.exec(t);
    if (m) return m[1];
    if (d === root || d === dirname(d)) return undefined;
  }
}

function readProjects(root) {
  const projects = [];
  for (const file of walkFiles(root)) {
    if (!file.endsWith(".csproj")) continue;
    // XML comments are ignored: a commented-out reference is not a reference.
    const text = readFileSync(file, "utf8").replace(/<!--[\s\S]*?-->/g, " ");
    const all = (re) => [...text.matchAll(re)].map((m) => m[1]);
    const name = basename(file, ".csproj");
    projects.push({
      name,
      file,
      dir: dirname(file),
      rel: posix(relative(root, file)),
      refs: all(/<ProjectReference\s[^>]*?Include\s*=\s*["']([^"']+)["']/gi)
        .map((r) => r.replace(/\$\((MSBuildThisFileDirectory|MSBuildProjectDirectory)\)[\\/]?/i, `${dirname(file)}/`))
        .filter((r) => !r.includes("$("))
        .map((r) => resolve(dirname(file), r.split("\\").join("/"))),
      packages: all(/<PackageReference\s[^>]*?Include\s*=\s*["']([^"']+)["']/gi),
      frameworks: all(/<FrameworkReference\s[^>]*?Include\s*=\s*"([^"]+)"/gi),
      ivt: all(/<InternalsVisibleTo\s[^>]*?Include\s*=\s*["']([^"']+)["']/gi).map((n) => n.split(",")[0].trim()),
      tfms: (/<TargetFrameworks?>([^<]+)</i.exec(text)?.[1] ?? inheritedTfm(file, root) ?? "").split(";").map((t) => t.trim()).filter(Boolean),
      isTest: /<IsTestProject>\s*true/i.test(text) || /Microsoft\.NET\.Test\.Sdk/.test(text) || /\.(Unit|Integration)?Tests?$/i.test(name),
    });
  }
  return detectModules(projects, root);
}

function detectModules(projects, root) {
  let config = {};
  try {
    config = JSON.parse(readFileSync(join(root, "modules.json"), "utf8"));
  } catch {
    // no config: use conventions
  }
  const stripRoles = (name) => {
    let base = name;
    for (let m = ROLE_SUFFIX.exec(base); m; m = ROLE_SUFFIX.exec(base)) base = base.slice(0, m.index);
    return base || name;
  };
  const bases = projects.map((p) => stripRoles(p.name));
  const multi = bases.filter((b) => b.includes("."));
  const first = multi.map((b) => b.split(".")[0]);
  const top = [...new Set(first)].map((s) => [s, first.filter((x) => x === s).length]).sort((a, b) => b[1] - a[1])[0];
  const prefix = top && multi.length >= 3 && top[1] / bases.length >= 0.8 ? `${top[0]}.` : "";
  // Path-boundary prefix match; the longest matching prefix wins (so "src/Orders" does not swallow "src/Orders2").
  const under = (rel, x) => {
    const base = x.replace(/\/+$/, "");
    return rel === base || rel.startsWith(`${base}/`);
  };
  const configured = (rel) => {
    const hits = [
      ...Object.entries(config.modules ?? {}).flatMap(([module, paths]) => paths.filter((x) => under(rel, x)).map((x) => ({ module, len: x.length }))),
      ...(config.shared ?? []).filter((x) => under(rel, x)).map((x) => ({ module: null, len: x.length })),
    ].sort((a, b) => b.len - a.len);
    return hits[0];
  };

  // When the repository clearly organises modules as Modules/<Name>/ (at least 4 projects), everything outside that folder
  // (hosts, building blocks, tools, cross-cutting tests) belongs to no module: a product or company name is not a module.
  const pathHit = (p) => /(?:^|\/)[Mm]odules\/([^/]+)\//.exec(`/${p.rel}`);
  const pathMode = !config.modules && projects.filter((p) => pathHit(p)).length >= 4;
  const fixed = new Set(); // projects whose module came from modules.json or the Modules/<Name> folder

  for (const [i, p] of projects.entries()) {
    let module;
    const hit = configured(p.rel);
    const m = pathHit(p);
    if (hit) module = hit.module;
    else if (m) module = m[1];
    else if (pathMode && !p.isTest) module = null; // tests fall through to their name, then to what they reference
    else if (SHARED_NAME.test(bases[i]) || HOST_NAME.test(bases[i])) module = null;
    // Acme.Api / Acme.Web: nothing left after the company prefix, so it is the host, not a module named "Acme".
    else if (prefix && bases[i] === prefix.slice(0, -1)) module = null;
    else module = prefix && bases[i].startsWith(prefix) ? bases[i].slice(prefix.length) : bases[i];
    if (hit || m) fixed.add(p);
    p.module = module;
    // "Zivex.Modules.Contracts" is the implementation of a module that is called Contracts; "Quotes.Contracts" is a contract project.
    const named = /\.Contracts$/i.test(p.name) && !(module === "Contracts" && !/\.Contracts\.Contracts$/i.test(p.name));
    p.role = p.isTest ? "test" : named ? "contracts" : "impl";
    if (module === null && !p.isTest) p.role = "shared";
  }

  // A test project belongs to the module its name points at; otherwise to the one module it references. One that references
  // several modules (architecture, integration, security suites) is cross-cutting and belongs to none.
  const byPath = new Map(projects.map((p) => [resolve(p.file), p]));
  const known = new Set(projects.filter((x) => !x.isTest && x.module).map((x) => x.module));
  for (const p of projects.filter((x) => x.isTest && !fixed.has(x))) {
    if (p.module && known.has(p.module)) continue; // the name points at a real module (Orders.Tests -> Orders): keep it
    const mods = new Set(p.refs.map((r) => byPath.get(r)).filter((t) => t && !t.isTest && t.module).map((t) => t.module));
    if (mods.size === 1) p.module = [...mods][0];
    else if (mods.size > 1 || pathMode) p.module = null;
  }
  return projects;
}

const projectOf = (projects, file) => projects.filter((p) => resolve(file).startsWith(`${p.dir}${process.platform === "win32" ? "\\" : "/"}`)).sort((a, b) => b.dir.length - a.dir.length)[0];
const byFile = (projects) => new Map(projects.map((p) => [resolve(p.file), p]));
const moduleNames = (projects) => [...new Set(projects.map((p) => p.module).filter(Boolean))].sort();
function tryRead(path) {
  try {
    return readFileSync(path, "utf8");
  } catch {
    return undefined;
  }
}

// Blank out /* ... */ comments but keep the newlines, so line numbers stay right.
const stripBlockComments = (text) => text.replace(/\/\*[\s\S]*?\*\//g, (m) => m.replace(/[^\n]/g, " "));
