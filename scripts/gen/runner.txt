// Heuristic, regex-based scan of a .NET repo (.cs and appsettings*.json). It finds candidates, not proof:
// read every hit in context before changing code. Skips bin/obj/node_modules/.git and test projects.
// Usage: node <this-file> [path] [--json] [--strict]    (--strict: exit 1 when any HIGH finding exists)
import { readdirSync, readFileSync } from "node:fs";
import { join, relative } from "node:path";

const args = process.argv.slice(2);
const root = args.find((a) => !a.startsWith("--")) ?? process.cwd();
const SKIP = new Set(["bin", "obj", "node_modules", ".git", ".vs", "TestResults"]);
const isTest = (rel) => /(^|[\\/])[^\\/]*\.?tests?([\\/]|\.)/i.test(rel);

function* walk(dir) {
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    if (e.isDirectory()) {
      if (!SKIP.has(e.name)) yield* walk(join(dir, e.name));
    } else if (/\.cs$|(^|[\\/])appsettings[^\\/]*\.json$/i.test(e.name)) yield join(dir, e.name);
  }
}
const lineOf = (text, index) => text.slice(0, index).split("\n").length;

const files = [];
for (const abs of walk(root)) {
  const rel = relative(root, abs).split("\\").join("/");
  if (!isTest(rel)) files.push({ rel, text: readFileSync(abs, "utf8"), cs: abs.endsWith(".cs") });
}

const findings = [];
const add = (r, file, line) => findings.push({ id: r.id, level: r.level, file, line, msg: r.msg, fix: r.fix });
for (const r of RULES) {
  if (r.project) {
    for (const hit of r.project(files)) add(r, hit.file, hit.line ?? 1);
    continue;
  }
  for (const f of files) {
    if (r.json ? f.cs : !f.cs) continue;
    if (r.re) {
      f.text.split("\n").forEach((l, i) => {
        if (f.cs && l.trimStart().startsWith("//")) return;
        if (r.re.test(l)) add(r, f.rel, i + 1);
      });
    } else if (r.multi) {
      for (const m of f.text.matchAll(r.multi)) add(r, f.rel, lineOf(f.text, m.index));
    } else if (r.fn) {
      for (const line of r.fn(f.text, f.rel)) add(r, f.rel, line);
    }
  }
}

const order = { high: 0, medium: 1, low: 2 };
findings.sort((a, b) => order[a.level] - order[b.level] || a.file.localeCompare(b.file) || a.line - b.line);
if (args.includes("--json")) console.log(JSON.stringify(findings, null, 2));
else {
  for (const f of findings) console.log(`${f.file}:${f.line}  ${f.level.toUpperCase()}  ${f.id}  ${f.msg}\n    fix: ${f.fix}`);
  const n = (l) => findings.filter((f) => f.level === l).length;
  console.log(`\n${files.length} files scanned: ${n("high")} high, ${n("medium")} medium, ${n("low")} low.`);
}
if (args.includes("--strict") && findings.some((f) => f.level === "high")) process.exit(1);
